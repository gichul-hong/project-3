# SD3.5-medium Subject-Driven Customization

**목표** CustomConcept101 10개 subject × 프롬프트 10개 = 100장 생성 후 CLIP-I / CLIP-T 채점.

**파이프라인**
1. `Pivotal Tuning` = Textual Inversion(신규 토큰 `<s0> <s1>`) + DreamBooth-LoRA 동시 학습
2. `Prior Preservation Loss`로 class 다양성 유지 (DreamBooth, Day2 p28)
3. 프롬프트당 후보를 과다 생성 → `CLIP 기반 자동 선별` (MMR 다양성 + 복붙 방지 가드)
4. (옵션) `RF-Inversion 레퍼런스 앵커링` — Day1 실습 scheduler 재사용
5. `evaluation.py` 동일 수식으로 22개 숫자(subject별 10 + 전체 1) × 2 산출

**실습 자료 재사용 영역 / 우리 아이디어 영역 구분**

| 구분 | 내용 |
|---|---|
| 실습(Day1) | `EulerInversion`, `ControlledODE`, `ControlledODEInversion`, `encode_image_to_sd3_latent` |
| 실습(Day2) | Textual Inversion 개념, DreamBooth 희귀토큰 + PPL, 어텐션 제어 이론 |
| **우리 아이디어 A** | SD3.5에 Pivotal Tuning 직접 구현 (diffusers SD3 스크립트에 없음) |
| **우리 아이디어 B** | 신규 토큰 임베딩의 class-noun 앵커 정규화 (드리프트 억제) |
| **우리 아이디어 C** | CLIP 기반 후보 선별 + MMR 다양성 + 근접복제 가드 (정성 기준 직접 최적화) |
| **우리 아이디어 D** | subject별 적응형 레시피 선택 + 통일/개별 점수 동시 보고 |
| **우리 아이디어 E** | LoRA와 RF-Inversion 앵커링 결합 (eta/tau로 CLIP-I↔CLIP-T 트레이드오프 제어) |

**실행 순서 권장** 1~4번 셀(환경) → 5~9번(데이터/유틸) → `PILOT` 2개 subject로 10~16번 검증 → `RUN_ALL`로 전체 sweep → 평가/제출.

# 6. zip 자동 탐색 → 압축 해제 → 정합성 점검
import zipfile

def find_zip(name_hint, explicit=""):
    """Drive/런타임에서 zip 을 자동 탐색. explicit 이 있으면 그것을 우선 사용."""
    if explicit and Path(explicit).exists():
        return Path(explicit)
    roots = [Path("/content/drive/MyDrive"), Path("/content"), Path("/content/drive/Shareddrives")]
    hits = []
    for r in roots:
        if not r.exists():
            continue
        try:
            hits += [p for p in r.rglob(f"*{name_hint}*.zip")
                     if p.is_file() and not any(x in p.parts for x in (".Trash", ".shortcut-targets-by-id"))]
        except Exception:
            pass
    if not hits:
        return None
    hits.sort(key=lambda p: (len(p.parts), -p.stat().st_size))
    if len(hits) > 1:
        print(f"  '{name_hint}' 후보 {len(hits)}개 → 선택: {hits[0]}")
    return hits[0]

def upload_fallback(name_hint):
    """Drive 에서 못 찾으면 브라우저 업로드로 대체 (브라우저 Colab 에서만 동작)."""
    try:
        from google.colab import files
    except Exception:
        return None
    print(f"[{name_hint}.zip] 을 찾지 못했습니다. 업로드 창을 엽니다...")
    up = files.upload()
    for fn in up:
        if fn.endswith(".zip") and name_hint.split(".")[0] in fn:
            dst = Path("/content") / fn
            print("  업로드 완료:", dst)
            return dst
    return None

def _clean_junk(root):
    """macOS 잔여물(__MACOSX, ._*, .DS_Store) 제거."""
    for d in [q for q in root.rglob("__MACOSX") if q.is_dir()]:
        shutil.rmtree(d, ignore_errors=True)
    for q in list(root.rglob("._*")) + list(root.rglob(".DS_Store")):
        if q.is_dir():
            shutil.rmtree(q, ignore_errors=True)
        elif q.exists():
            q.unlink()

def _flatten_if_wrapped(root, expect):
    """zip 안에 폴더가 한 겹 더 있을 때만 평탄화.

    expect 중 하나라도 root 바로 아래에 있으면 이미 올바른 구조이므로 건드리지 않는다.
    """
    if any((root / e).exists() for e in expect):
        return
    subdirs = [q for q in root.iterdir() if q.is_dir()]
    if len(subdirs) != 1:
        return
    inner = subdirs[0]
    if not any((inner / e).exists() for e in expect):
        return
    print(f"  한 겹 감싸인 구조 감지 → 평탄화: {inner.name}/")
    for q in list(inner.iterdir()):
        shutil.move(str(q), str(root / q.name))
    shutil.rmtree(inner, ignore_errors=True)

def unzip(src, dst, expect, force=False):
    """압축 해제 + 잔여물 제거 + 구조 검증. 구조가 어긋나면 자동으로 다시 푼다."""
    dst = Path(dst)
    have = dst.exists() and any((dst / e).exists() for e in expect)
    if have and not force:
        print("skip (already valid):", dst); return
    if dst.exists():
        shutil.rmtree(dst, ignore_errors=True)   # 깨진 이전 상태 폐기
    dst.mkdir(parents=True, exist_ok=True)

    with zipfile.ZipFile(src) as z:
        z.extractall(dst)
    _clean_junk(dst)
    _flatten_if_wrapped(dst, expect)

    missing = [e for e in expect if not (dst / e).exists()]
    if missing:
        print(f"  [경고] 다음 항목을 찾지 못했습니다: {missing[:5]}")
        print(f"  실제 최상위 내용: {sorted(q.name for q in dst.iterdir())[:12]}")
    else:
        print("extracted & verified:", dst)

data_zip = find_zip("dataset", CFG["DATA_ZIP"]) or upload_fallback("dataset")
prom_zip = find_zip("prompt",  CFG["PROMPT_ZIP"]) or upload_fallback("prompt")
assert data_zip, "dataset.zip 을 찾을 수 없습니다. CFG['DATA_ZIP'] 에 경로를 직접 지정하세요."
assert prom_zip, "prompt.zip 을 찾을 수 없습니다. CFG['PROMPT_ZIP'] 에 경로를 직접 지정하세요."
print("dataset.zip :", data_zip, f"({data_zip.stat().st_size/1e6:.0f} MB)")
print("prompt.zip  :", prom_zip)

# 구조가 깨져 있으면 자동으로 다시 풀도록 기대 항목을 넘긴다.
FORCE_REEXTRACT = False   # 문제가 계속되면 True 로 한 번 실행

unzip(data_zip, DATA, expect=SUBJECTS, force=FORCE_REEXTRACT)
unzip(prom_zip, PROM, expect=[f"{s}.txt" for s in SUBJECTS], force=FORCE_REEXTRACT)

IMG_EXT = {".png", ".jpg", ".jpeg", ".webp", ".bmp"}

def ref_paths(subject):
    d = DATA / subject
    return sorted([p for p in d.iterdir()
                   if p.suffix.lower() in IMG_EXT and not p.name.startswith("._")])

def eval_prompts(subject):
    # evaluation.py 와 동일하게 {} -> class noun 치환 (채점용, 수정 금지)
    f = PROM / f"{subject}.txt"
    return [l.strip().replace("{}", CLASS_PROMPT[subject])
            for l in f.read_text(encoding="utf-8").splitlines() if l.strip()]

print(f"\n{'subject':24s} refs  prompts")
ok = True
for s in SUBJECTS:
    d = DATA / s
    if not d.exists():
        print(f"{s:24s}  --  폴더 없음"); ok = False; continue
    n, m = len(ref_paths(s)), len(eval_prompts(s))
    ok &= (n > 0 and m == 10)
    print(f"{s:24s} {n:4d}  {m:5d}")
print("\n데이터 정합성:", "OK" if ok else "확인 필요 (프롬프트 10줄 / 레퍼런스 1장 이상)")

if not ok:
    print("\n--- 진단 정보 ---")
    print("DATA:", DATA)
    print("  최상위:", sorted(q.name for q in DATA.iterdir())[:15] if DATA.exists() else "없음")
    print("PROM:", PROM)
    print("  최상위:", sorted(q.name for q in PROM.iterdir())[:15] if PROM.exists() else "없음")
    print("\n위 목록에 subject 폴더가 없다면 FORCE_REEXTRACT = True 로 바꿔 이 셀을 다시 실행하세요.")

`evaluation.py`와 **동일한 수식**을 쓰되, `transformers` 버전에 따라 `get_text_features().pooler_output`이
동작하지 않는 경우가 있어 호환 래퍼를 둡니다. (원본 파일은 수정하지 않습니다.)

- **CLIP-T** i번째 생성 이미지 ↔ i번째 프롬프트, 그 평균
- **CLIP-I** 생성 이미지 × 레퍼런스 전체 쌍 코사인 유사도 평균

============================== CELL 10 (code) ==============================
# 7. CLIP 로드 및 채점 함수 (evaluation.py 호환)
import torch, torch.nn.functional as F
from PIL import Image
from transformers import CLIPModel, CLIPProcessor

CLIP_ID = "openai/clip-vit-base-patch32"
_dev = "cuda" if torch.cuda.is_available() else "cpu"
clip_model = CLIPModel.from_pretrained(CLIP_ID).to(_dev).eval()
clip_proc  = CLIPProcessor.from_pretrained(CLIP_ID)

def _unwrap(x):
    # transformers 버전차 흡수: 텐서면 그대로, 출력객체면 pooler_output
    return x if isinstance(x, torch.Tensor) else x.pooler_output

@torch.no_grad()
def clip_text_emb(prompts):
    b = clip_proc(text=prompts, return_tensors="pt", padding=True, truncation=True).to(_dev)
    return _unwrap(clip_model.get_text_features(**b)).float()

@torch.no_grad()
def clip_image_emb(images, bs=32):
    out = []
    for i in range(0, len(images), bs):
        b = clip_proc(images=images[i:i + bs], return_tensors="pt").to(_dev)
        out.append(_unwrap(clip_model.get_image_features(**b)).float())
    return torch.cat(out)

def load_rgb(p):
    return Image.open(p).convert("RGB")

@torch.no_grad()
def score_subject(subject, image_dir):
    """evaluation.py 와 동일한 CLIP-T / CLIP-I 계산."""
    gen = sorted(Path(image_dir).glob("*"))
    gen = [p for p in gen if p.suffix.lower() in IMG_EXT]
    prompts = eval_prompts(subject)
    gi = clip_image_emb([load_rgb(p) for p in gen])
    ri = clip_image_emb([load_rgb(p) for p in ref_paths(subject)])
    te = clip_text_emb(prompts)
    n = min(len(gi), len(te))
    clip_t = F.cosine_similarity(gi[:n], te[:n]).mean().item()
    clip_i = F.cosine_similarity(gi.unsqueeze(1), ri.unsqueeze(0), dim=-1).mean().item()
    return dict(subject=subject, clip_t=clip_t, clip_i=clip_i, n_gen=len(gen), n_ref=len(ri))

print("CLIP ready on", _dev)

        ).images
        for im in imgs:
            im.save(out / f"{len(have) + made:04d}.jpg", quality=95)
            made += 1
    flush()
    return sorted(out.glob("*.jpg"))[:n]

## Pivotal Tuning 학습

**이 섹션이 우리 아이디어의 핵심입니다.** diffusers의 SD3 DreamBooth-LoRA 스크립트에는
Textual Inversion 옵션이 없으므로(SDXL advanced 스크립트에만 존재) 직접 구현합니다.

동시에 학습하는 두 대상:
1. **신규 토큰 임베딩** `<s0> <s1>` — CLIP-L / CLIP-G 두 인코더 각각에 추가. Textual Inversion(Day2 p7)
2. **MMDiT LoRA 어댑터** — attention의 `to_q/k/v/out` 에 rank 32. DreamBooth를 LoRA로 경량화

추가 장치:
- **아이디어 B: 앵커 정규화** 신규 임베딩이 class noun 임베딩에서 과도하게 멀어지지 않게 L2로 당깁니다.
  임베딩 폭주(norm explosion)로 프롬프트 추종이 깨지는 TI의 대표 실패를 억제합니다.
- **PPL** class 이미지 배치를 함께 흘려 prior를 보존합니다.
- LoRA 학습 시 텍스트 인코더는 **동결**하고 임베딩 행렬만 학습합니다(학습량 최소화).

# 11. 신규 토큰 추가 + 임베딩 초기화 (Textual Inversion 파트)
import torch.nn as nn

NEW_TOKENS = [f"<s{i}>" for i in range(CFG["NUM_NEW_TOKENS"])]

def setup_tokens(subject):
    """두 텍스트 인코더에 신규 토큰을 추가하고 class noun 임베딩 평균으로 초기화.

    여러 subject 를 연속 학습해도 토큰이 누적되지 않도록 멱등하게 동작한다.
    """
    tokenizers    = [pipe.tokenizer, pipe.tokenizer_2]
    text_encoders = [pipe.text_encoder, pipe.text_encoder_2]
    class_noun = CLASS_PROMPT[subject]
    handles = []

    for tok, te in zip(tokenizers, text_encoders):
        unk = tok.convert_tokens_to_ids(NEW_TOKENS[0])
        already = unk is not None and unk != tok.unk_token_id
        if not already:
            tok.add_tokens(NEW_TOKENS)
            te.resize_token_embeddings(len(tok))
        emb = te.get_input_embeddings()
        ids = tok.convert_tokens_to_ids(NEW_TOKENS)

        # class noun 의 서브워드 임베딩 평균으로 초기화 → 학습 초반 안정화
        base_ids = tok(class_noun, add_special_tokens=False).input_ids
        with torch.no_grad():
            init = emb.weight[base_ids].mean(dim=0, keepdim=True).clone()
            for j, i in enumerate(ids):
                noise = torch.randn_like(init) * 0.01 * init.norm() / (init.numel() ** 0.5)
                emb.weight[i] = (init + noise).squeeze(0)
        handles.append(dict(tokenizer=tok, encoder=te, ids=ids,
                            anchor=emb.weight[base_ids].mean(dim=0).detach().clone()))
    return handles

def trainable_embedding_params(handles):
    ps = []
    for h in handles:
        emb = h["encoder"].get_input_embeddings()
        emb.weight.requires_grad_(True)
        ps.append(emb.weight)
    return ps

def freeze_non_embedding(handles):
    """임베딩 행렬 외 텍스트 인코더 전체 동결."""
    for h in handles:
        for n, p in h["encoder"].named_parameters():
            p.requires_grad_("embeddings.token_embedding" in n or n.endswith("shared.weight"))

def zero_grad_except_new(handles):
    """임베딩 행렬 중 신규 토큰 행만 남기고 grad 마스킹 → 기존 vocab 보존."""
    for h in handles:
        emb = h["encoder"].get_input_embeddings()
        if emb.weight.grad is None:
            continue
        mask = torch.zeros_like(emb.weight.grad)
        mask[h["ids"]] = 1.0
        emb.weight.grad *= mask

def anchor_loss(handles):
    """아이디어 B: 신규 임베딩을 class noun 앵커 쪽으로 당기는 정규화."""
    loss = 0.0
    for h in handles:
        emb = h["encoder"].get_input_embeddings()
        v = emb.weight[h["ids"]]
        a = h["anchor"].to(v.device, v.dtype).unsqueeze(0)
        loss = loss + ((v - a) ** 2).mean()
    return loss

# 12. 학습 데이터셋 (레퍼런스 + class 이미지)
from torch.utils.data import Dataset, DataLoader
from torchvision import transforms as T
from PIL import ImageOps

def fit_square(im, res, mode="pad", fill=(255, 255, 255)):
    """정사각형으로 맞추되 subject 가 잘리지 않게 한다.

    Resize+CenterCrop 은 종횡비가 1 에서 멀 때 위/아래를 잘라낸다.
    예: scene_waterfall 은 종횡비 0.65 (세로로 긴 사진) 라서 center crop 시
        정체성의 핵심인 아치형 다리나 폭포 상단이 사라진다.

    mode="pad"  : 전체를 담고 남는 부분을 채운다 (정보 손실 없음)
    mode="crop" : 기존 동작 (중앙을 잘라낸다)
    """
    if mode == "crop":
        w, h = im.size
        k = res / min(w, h)
        im = im.resize((max(res, int(round(w * k))), max(res, int(round(h * k)))),
                       Image.BICUBIC)
        w, h = im.size
        l, t = (w - res) // 2, (h - res) // 2
        return im.crop((l, t, l + res, t + res))

    # pad: 긴 변을 res 에 맞추고 짧은 변을 채워 정사각형으로
    im = ImageOps.contain(im, (res, res), Image.BICUBIC)
    canvas = Image.new("RGB", (res, res), fill)
    canvas.paste(im, ((res - im.width) // 2, (res - im.height) // 2))
    return canvas

# 종횡비가 1 에서 벗어난 이미지를 center crop 하면 정체성 일부가 잘려나간다.
# 실측 손실면적: furniture_sofa2 25%(L자 소파 형태 손상), transport_tank 3.7%,
#               scene_waterfall 은 세로 사진(0.65~0.75)이라 아치형 다리가 잘릴 위험.
# 따라서 subject 목록을 고정하지 않고 실제 종횡비로 자동 판정한다.
AR_TOLERANCE = 0.12          # |AR-1| 이 이 값을 넘으면 pad 사용
FORCE_PAD  = {"scene_waterfall", "furniture_sofa2", "actionfigure_2", "person_3"}
FORCE_CROP = set()

def _median_ar(subject):
    ars = []
    for p in ref_paths(subject):
        try:
            with Image.open(p) as im:
                w, h = im.size
            ars.append(w / h)
        except Exception:
            pass
    if not ars:
        return 1.0
    ars.sort()
    return ars[len(ars) // 2]

_FIT_CACHE = {}

def fit_mode_for(subject):
    """레퍼런스 종횡비를 보고 pad/crop 을 자동 결정 (결과 캐시)."""
    if subject in _FIT_CACHE:
        return _FIT_CACHE[subject]
    if subject in FORCE_CROP:
        m = "crop"
    elif subject in FORCE_PAD:
        m = "pad"
    else:
        m = "pad" if abs(_median_ar(subject) - 1.0) > AR_TOLERANCE else "crop"
    _FIT_CACHE[subject] = m
    return m

        self.tf = T.Compose([

    def _load(self, p, mode=None):

        im = fit_square(im, self.res, mode or self.fit_mode)
        if self.flip and random.random() < self.flip:

    def __getitem__(self, i):

        pixel=torch.stack([b["pixel"] for b in batch]),

# 14. Pivotal Tuning 학습 루프 (LoRA + 임베딩 + PPL + 앵커 정규화)
from tqdm.auto import tqdm

def encode_prompts(prompts):
    """신규 토큰이 반영된 상태로 프롬프트 인코딩. 임베딩 학습을 위해 grad 유지."""
    pe, _neg, pooled, _neg_pooled = pipe.encode_prompt(
        prompt=prompts, prompt_2=prompts, prompt_3=None,
        do_classifier_free_guidance=False,
        max_sequence_length=CFG["MAX_SEQ"],
        device=DEVICE,
    )
    return pe, pooled

@torch.no_grad()
def vae_encode(px):
    px = px.to(DEVICE, dtype=pipe.vae.dtype)
    lat = pipe.vae.encode(px).latent_dist.sample()
    return (lat - VAE_SH) * VAE_SF

def flow_match_loss(latents, pe, pooled, gen):
    """SD3 rectified flow 학습 목표: v = noise - x0 (Day1 RF 배경 p39~40)."""
    bsz = latents.shape[0]
    noise = torch.randn(latents.shape, device=DEVICE, dtype=latents.dtype, generator=gen)
    # logit-normal sigma 샘플링 (SD3 논문 기본). rectified flow 에서는 sigma == t.
    n_train = pipe.scheduler.config.num_train_timesteps
    u = torch.normal(0.0, 1.0, size=(bsz,), device=DEVICE).sigmoid().clamp(1e-3, 1 - 1e-3)
    sigmas = u.view(-1, 1, 1, 1).to(latents.dtype)
    noisy = (1.0 - sigmas) * latents + sigmas * noise
    timesteps = (u * n_train).to(latents.dtype)

    pred = pipe.transformer(
        hidden_states=noisy,
        timestep=timesteps,
        encoder_hidden_states=pe,
        pooled_projections=pooled,
        return_dict=False,
    )[0]
    target = noise - latents
    return F.mse_loss(pred.float(), target.float())

def train_subject(subject, steps=None, use_ppl=None, lr_lora=None, lr_emb=None,
                  anchor_w=None, tag="", verbose=True):
    steps    = steps    if steps    is not None else CFG["TRAIN_STEPS"]
    use_ppl  = use_ppl  if use_ppl  is not None else CFG["USE_PPL"]
    lr_lora  = lr_lora  if lr_lora  is not None else CFG["LR_LORA"]
    lr_emb   = lr_emb   if lr_emb   is not None else CFG["LR_EMB"]
    anchor_w = anchor_w if anchor_w is not None else CFG["EMB_ANCHOR_W"]

    t0 = time.time()
    print(f"[train] {subject} steps={steps} ppl={use_ppl} {tag}")
    class_imgs = ensure_class_images(subject) if use_ppl else []

    pipe.scheduler = FlowMatchEulerDiscreteScheduler.from_config(BASE_SCHED_CFG)
    handles = setup_tokens(subject)
    freeze_non_embedding(handles)
    pipe.vae.requires_grad_(False)
    pipe.transformer.requires_grad_(False)
    lora_params = attach_lora()
    emb_params  = trainable_embedding_params(handles)

    if CFG["GRAD_CKPT"]:
        pipe.transformer.enable_gradient_checkpointing()
    pipe.transformer.train()

    opt = torch.optim.AdamW(
        [dict(params=lora_params, lr=lr_lora, weight_decay=1e-4),
         dict(params=emb_params,  lr=lr_emb,  weight_decay=0.0)],
        betas=(0.9, 0.999), eps=1e-8,
    )
    sched = torch.optim.lr_scheduler.OneCycleLR(
        opt, max_lr=[lr_lora, lr_emb], total_steps=steps, pct_start=0.1, anneal_strategy="cos")

    ds = SubjectDataset(subject, CFG["TRAIN_RES"], class_imgs)
    dl = DataLoader(ds, batch_size=1, shuffle=True, num_workers=2, collate_fn=collate, drop_last=True)
    gen = torch.Generator(device=DEVICE).manual_seed(CFG["SEED"])

    it = iter(dl)
    pbar = tqdm(range(steps), disable=not verbose)
    for step in pbar:
        try:
            batch = next(it)
        except StopIteration:
            it = iter(dl); batch = next(it)

        lat = vae_encode(batch["pixel"])
        pe, pooled = encode_prompts(batch["prompt"])
        loss = flow_match_loss(lat, pe, pooled, gen)

        if use_ppl and "class_pixel" in batch:
            clat = vae_encode(batch["class_pixel"])
            cpe, cpooled = encode_prompts(batch["class_prompt"])
            loss = loss + CFG["PPL_WEIGHT"] * flow_match_loss(clat, cpe, cpooled, gen)

        if anchor_w > 0:
            loss = loss + anchor_w * anchor_loss(handles)

        loss.backward()
        zero_grad_except_new(handles)
        torch.nn.utils.clip_grad_norm_(lora_params + emb_params, 1.0)
        opt.step(); sched.step(); opt.zero_grad(set_to_none=True)

        if verbose and step % 50 == 0:
            pbar.set_postfix(loss=f"{loss.item():.4f}")

    pipe.transformer.eval()
    if CFG["GRAD_CKPT"]:
        pipe.transformer.disable_gradient_checkpointing()
    save_pivotal(subject, handles)
    flush()
    print(f"[train] {subject} done in {(time.time()-t0)/60:.1f} min")
    return handles

# 15. Day1 실습 scheduler 재사용 (RF-Inversion)
import numpy as np
from diffusers.schedulers.scheduling_flow_match_euler_discrete import FlowMatchEulerDiscreteSchedulerOutput

class EulerInversion(FlowMatchEulerDiscreteScheduler):
    """[실습 Day1] FlowMatch Euler schedule을 data→noise 방향으로 뒤집는다."""
    def set_timesteps(self, num_inference_steps=None, device=None, sigmas=None, mu=None, timesteps=None):
        super().set_timesteps(num_inference_steps=num_inference_steps, device=device,
                              sigmas=sigmas, mu=mu, timesteps=timesteps)
        self.timesteps = torch.flip(self.timesteps, dims=(0,))
        self.sigmas = torch.flip(self.sigmas, dims=(0,))
        self._step_index = None
        self._begin_index = None

class ControlledODE(FlowMatchEulerDiscreteScheduler):
    """[실습 Day1 + Todo 구현] 원본 latent 방향 conditional velocity를 섞는 생성 scheduler."""
    def set(self, reference, tau=0.7, eta=0.9):
        self.reference = reference; self.tau = float(tau); self.eta = float(eta); return self

    def controller(self, sample, sigma):
        reference = self.reference.to(device=sample.device, dtype=sample.dtype)
        # 생성 방향 conditional velocity: v(x_t | x_0) = (x_t - x_0) / t,  t = sigma
        return (sample - reference) / max(float(sigma), 1e-3)

    def step(self, model_output, timestep, sample, *args, return_dict=True, **kwargs):
        if self.step_index is None:
            self._init_step_index(timestep)
        sample = sample.to(torch.float32)
        sigma = self.sigmas[self.step_index].to(sample.device)
        sigma_next = self.sigmas[self.step_index + 1].to(sample.device)
        conditional_velocity = self.controller(sample, sigma).to(model_output.dtype)
        eta = self.eta if sigma.item() > self.tau else 0.0
        controlled_velocity = model_output + eta * (conditional_velocity - model_output)
        prev_sample = (sample + (sigma_next - sigma) * controlled_velocity).to(model_output.dtype)
        self._step_index += 1
        if not return_dict:
            return (prev_sample,)
        return FlowMatchEulerDiscreteSchedulerOutput(prev_sample=prev_sample)

class ControlledODEInversion(ControlledODE):
    """[실습 Day1 + Todo 구현] sampled prior 방향으로 섞는 inversion scheduler."""
    def set(self, reference, tau=0.0, eta=0.5):
        return super().set(reference, tau=tau, eta=eta)

    def controller(self, sample, sigma):
        reference = self.reference.to(device=sample.device, dtype=sample.dtype)
        # inversion 방향 conditional velocity: v(x_t | z) = (z - x_t) / (1 - t),  t = sigma
        return (reference - sample) / max(1.0 - float(sigma), 1e-3)

    def set_timesteps(self, num_inference_steps=None, device=None, sigmas=None, mu=None, timesteps=None):
        FlowMatchEulerDiscreteScheduler.set_timesteps(
            self, num_inference_steps=num_inference_steps, device=device,
            sigmas=sigmas, mu=mu, timesteps=timesteps)
        self.timesteps = torch.flip(self.timesteps, dims=(0,))
        self.sigmas = torch.flip(self.sigmas, dims=(0,))
        self._step_index = None
        self._begin_index = None

@torch.no_grad()
def encode_image_to_sd3_latent(image, seed=0):
    """[실습 Day1] 이미지를 SD3 latent로 변환."""
    px = pipe.image_processor.preprocess(image).to(device=DEVICE, dtype=pipe.vae.dtype)
    post = pipe.vae.encode(px).latent_dist
    raw = post.sample(generator=torch.Generator(device=DEVICE).manual_seed(seed))
    return (raw - VAE_SH) * VAE_SF

@torch.no_grad()
def rf_anchor_latent(subject, ref_idx=0, tau=0.0, eta=0.5, steps=None):
    """레퍼런스 이미지를 RF-Inversion 해서 생성 초기 latent(앵커)로 만든다."""
    steps = steps or CFG["STEPS"]
    # 학습과 동일한 방식으로 정사각화 (단순 resize 는 종횡비를 왜곡시킨다)
    im = fit_square(load_rgb(ref_paths(subject)[ref_idx]), CFG["GEN_RES"], fit_mode_for(subject))
    lat = encode_image_to_sd3_latent(im, seed=CFG["SEED"])
    prior = torch.randn(lat.shape, generator=torch.Generator(device=DEVICE).manual_seed(CFG["SEED"]),
                        device=lat.device, dtype=lat.dtype)
    sch = ControlledODEInversion.from_config(BASE_SCHED_CFG).set(prior, tau=tau, eta=eta)
    pipe.scheduler = sch
    inv = pipe(prompt="", num_inference_steps=steps,
               height=CFG["GEN_RES"], width=CFG["GEN_RES"],
               guidance_scale=1.0, latents=lat, output_type="latent",
               max_sequence_length=CFG["MAX_SEQ"]).images.detach()
    pipe.scheduler = FlowMatchEulerDiscreteScheduler.from_config(BASE_SCHED_CFG)
    flush()
    return inv

def blend_anchor(anchor, seed, strength=0.85):
    """앵커 latent 에 seed별 노이즈를 섞어 후보 다양성을 확보한다.

    strength=1.0 이면 순수 노이즈(앵커 무시), 0.0 이면 앵커 그대로(모든 후보 동일).
    SDEdit(Day1) 의 realism-faithfulness 트레이드오프를 latent 초기화에 적용한 형태.
    """
    g = torch.Generator(device=DEVICE).manual_seed(seed)
    noise = torch.randn(anchor.shape, generator=g, device=anchor.device, dtype=anchor.dtype)
    return (1.0 - strength) * anchor + strength * noise

## 후보 과다생성 + CLIP 자동 선별 (아이디어 C)

```
score = W_CLIP_T·cos(img, prompt) + W_CLIP_I·mean cos(img, refs)
        − W_DIVERSE·max cos(img, 이미 선택된 이미지들)      ← 획일성 방지
```

앞 두 항은 정량(CLIP-T/CLIP-I), 뒤 두 항은 **정성 평가 기준을 직접 목적함수에 넣은 것**입니다.

@torch.no_grad()
def select_best(subject, candidates, out_dir=None, w_t=None, w_i=None,
                w_div=None, dup_th=None, dup_pen=None, report=True, run=None):
    """MMR + 복붙가드 기반 선별 후 0.png..9.png 저장."""

    w_i     = w_i     if w_i     is not None else CFG["W_CLIP_I"]
    w_div   = w_div   if w_div   is not None else CFG["W_DIVERSE"]
    dup_th  = dup_th  if dup_th  is not None else CFG["DUP_THRESH"]
    dup_pen = dup_pen if dup_pen is not None else CFG["DUP_PENALTY"]

    run = run or "adhoc"
    out_dir = Path(out_dir or run_dir(subject, run, "images"))
    out_dir.mkdir(parents=True, exist_ok=True)
    for p in out_dir.glob("*.png"):
        p.unlink()

    prompts = eval_prompts(subject)          # 채점과 동일한 원본 프롬프트로 선별
    te = F.normalize(clip_text_emb(prompts), dim=-1)
    ri = F.normalize(clip_image_emb([load_rgb(p) for p in ref_paths(subject)]), dim=-1)

    chosen_emb, rows = [], []
    for pi in range(10):
        imgs = candidates[pi]
        gi = F.normalize(clip_image_emb(imgs), dim=-1)
        s_t = gi @ te[pi]                        # CLIP-T
        s_i = (gi @ ri.T).mean(dim=1)            # CLIP-I
        s_dup = (gi @ ri.T).max(dim=1).values    # 최근접 레퍼런스
        score = w_t * s_t + w_i * s_i - dup_pen * (s_dup - dup_th).clamp(min=0)
        if chosen_emb:
            prev = torch.stack(chosen_emb)
            score = score - w_div * (gi @ prev.T).max(dim=1).values
        b = int(score.argmax())
        chosen_emb.append(gi[b])
        imgs[b].save(out_dir / f"{pi}.png")
        rows.append(dict(prompt_idx=pi, picked=b, clip_t=float(s_t[b]),
                         clip_i=float(s_i[b]), dup=float(s_dup[b]), score=float(score[b])))
        if report:
            print(f"  p{pi}: pick c{b}  T={s_t[b]:.4f} I={s_i[b]:.4f} dup={s_dup[b]:.4f}")
    (out_dir / "selection.json").write_text(json.dumps(rows, indent=2), encoding="utf-8")
    # 최근 실행 포인터 (제출/평가 시 어떤 run 을 쓸지 결정)
    (DR / "images" / subject / "LATEST.txt").write_text(run, encoding="utf-8")
    return rows

# 17. 파일럿: 1개 subject 빠른 검증 (약 12분)
def run_one(subject, train_steps=None, n_cand=None, use_rf=False,
            rf_tau=0.6, rf_eta=0.4, rf_strength=0.85, skip_train=False,
            tag="", run=None):
    """학습 → 후보생성 → 선별 → 채점을 한 subject에 대해 수행.

    결과는 images/<subject>/<run>/ 아래 보관되므로 이전 실험이 덮어써지지 않는다.
    """
    t0 = time.time()
    run = run or make_run_name(tag=tag,
                               train_steps=train_steps or CFG["TRAIN_STEPS"],
                               n_cand=n_cand or CFG["CANDIDATES"],
                               use_rf=use_rf, rf_eta=rf_eta)
    print(f"[run] {subject} / {run}")

    detach_lora()
    lora_f = DR / "lora" / subject / "pytorch_lora_weights.safetensors"
    if skip_train and lora_f.exists():
        print(f"[load] {subject} 기존 LoRA 재사용")
        load_pivotal(subject)
    else:
        train_subject(subject, steps=train_steps, tag=tag)

    anchor = rf_anchor_latent(subject, tau=rf_tau, eta=rf_eta) if use_rf else None
    cands = generate_candidates(subject, n_cand=n_cand, anchor=anchor,
                                anchor_strength=rf_strength, run=run)
    select_best(subject, cands, run=run)

    out = run_dir(subject, run, "images")
    sc = score_subject(subject, out)
    sc.update(minutes=round((time.time() - t0) / 60, 1),
              tag=tag or ("rf" if use_rf else "base"),
              run=run, path=str(out),
              train_steps=train_steps or CFG["TRAIN_STEPS"],
              n_cand=n_cand or CFG["CANDIDATES"], use_rf=bool(use_rf))
    print(f"[score] {subject}  CLIP-T={sc['clip_t']:.4f}  "
          f"CLIP-I={sc['clip_i']:.4f}  ({sc['minutes']}min)")
    print(f"        저장: {out}")

    # 실험 이력 누적 (덮어쓰지 않고 append)
    hist_f = DR / "scores" / "history.json"
    hist = json.loads(hist_f.read_text(encoding="utf-8")) if hist_f.exists() else []
    hist.append(sc)
    hist_f.write_text(json.dumps(hist, indent=2, ensure_ascii=False), encoding="utf-8")
    return sc

def pick_run(subject, mode="latest"):
    """제출에 쓸 run 을 고른다.

    mode="latest" : LATEST.txt / 최신 폴더
    mode="best"   : history 에서 CLIP-T + CLIP-I 합이 최대인 run

# 24. 제출 zip 생성 + 무결성 검증
def build_submission(name="submission_images"):
    stage = W / name
    if stage.exists():
        shutil.rmtree(stage)
    stage.mkdir(parents=True)

    problems = []
    total = 0
    used = {}
    for s in SUBJECTS:
        r = pick_run(s, SELECT_MODE)
        used[s] = r
        src, dst = run_dir(s, r, "images"), stage / s
        dst.mkdir(parents=True)
        for i in range(10):
            p = src / f"{i}.png"
            if not p.exists():
                problems.append(f"{s}/{i}.png 없음"); continue
            shutil.copy2(p, dst / f"{i}.png"); total += 1
        # 정렬 순서 == 프롬프트 순서 확인
        names = sorted(q.name for q in dst.glob("*.png"))
        if names != [f"{i}.png" for i in range(10)]:
            problems.append(f"{s}: 파일명/정렬 불일치 {names}")

    zip_path = DR / "submit" / name
    shutil.make_archive(str(zip_path), "zip", root_dir=stage)
    (DR / "submit" / "used_runs.json").write_text(
        json.dumps(used, indent=2, ensure_ascii=False), encoding="utf-8")
    print(f"이미지 {total}/100장 → {zip_path}.zip")
    print("\n사용된 run:")
    for k, v in used.items():
        print(f"  {k:22s} {v}")
    if problems:
        print("\n[문제]")
        for p in problems:
            print(" -", p)
    else:
        print("검증 통과: 100장, 파일명/순서 정상")
    return f"{zip_path}.zip"

build_submission()

# 26. 통일 적용 vs 개별 적용 비교표 (조교 요구사항)
def compare_uniform_vs_adaptive():
    f = DR / "scores" / "all_scores.json"
    if not f.exists():
        print("all_scores.json 없음 (전체 sweep 먼저 실행)"); return
    data = json.loads(f.read_text(encoding="utf-8"))
    groups = {}
    for s, r in data.items():
        groups.setdefault(r.get("tag", "unknown"), []).append(r)
    print(f"{'설정':12s} {'subjects':>9s} {'CLIP-T':>9s} {'CLIP-I':>9s}")
    print("-" * 44)
    for tag, rows in groups.items():
        mt = sum(r["clip_t"] for r in rows) / len(rows)
        mi = sum(r["clip_i"] for r in rows) / len(rows)
        print(f"{tag:12s} {len(rows):9d} {mt:9.4f} {mi:9.4f}")
    print("\nsubject별 적용 방법:")
    for s in SUBJECTS:
        if s in data:
            rc = RECIPE[s]
            method = "LoRA+PT" + (" +RF-Inv" if rc.get("use_rf") else "")
            print(f"  {s:24s} {method:18s} steps={rc['train_steps']}")

## 실험 이력 관리

모든 실행 결과는 덮어써지지 않고 아래 구조로 쌓입니다.

```
sd35_project/
  images/<subject>/<run>/0.png ... 9.png   # run = 20250819_1432__pilot__st1600_c4_rfoff
  images/<subject>/<run>/selection.json    # 후보 선별 근거 (T/I/dup 값)
  images/<subject>/LATEST.txt              # 최근 run 포인터
  candidates/<subject>/<run>/              # 후보 전량
  scores/history.json                      # 전체 실험 이력
  submit/used_runs.json                    # 제출에 사용한 run 기록
```

유용한 명령
- `show_history()` — 모든 실험 비교표
- `show_history("pet_cat5")` — 특정 subject 이력만
- `show_grid("pet_cat5")` — 최근 run 그리드
- `show_grid("pet_cat5", run="20250819_1432__pilot__st1600_c4_rfoff")` — 특정 run 지정
- `list_runs("pet_cat5")` — 해당 subject 의 run 목록

`SELECT_MODE = "best"` 이면 제출·채점 시 **subject 별로 점수가 가장 좋은 run** 을 자동 선택합니다.
`"latest"` 로 바꾸면 가장 최근 run 을 씁니다.
